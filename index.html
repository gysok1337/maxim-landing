var heroLines=gsap.utils.toArray('.hero .line-in'), heroRift=document.querySelector('.code-rift');
var heroRiftWell=document.querySelector('.code-rift-well'), heroRiftPlane=document.querySelector('.code-rift-plane');
var heroRiftStream=document.querySelector('.code-rift-stream'), heroSection=document.querySelector('.hero');

gsap.set(heroLines,{yPercent:120});
gsap.set(heroRiftWell,{scaleX:.08});
gsap.set(heroRiftStream,{x:-46,opacity:.16});

function heroIn(delay){
  var timeline=gsap.timeline({delay:delay||0});
  timeline.to(heroLines,{yPercent:0,duration:1.15,ease:'power4.out',stagger:.09})
    .to(heroRift,{opacity:1,x:0,y:0,scale:1,duration:1.08,ease:'power4.out'},'-=.94')
    .to(heroRiftWell,{scaleX:1,duration:1.02,ease:'power4.inOut'},'<')
    .to(heroRiftStream,{x:0,opacity:1,duration:1.1,ease:'power3.out'},'<.18');
  return timeline;
}

// the lower layer stays fixed while the surface moves
var riftMedia=gsap.matchMedia();
riftMedia.add('(min-width: 761px)',function(){
  if(!heroSection||!heroRiftPlane)return;
  gsap.fromTo(heroRiftPlane,{y:0},{
    y:function(){return heroSection.offsetHeight;},ease:'none',
    scrollTrigger:{trigger:heroSection,start:'top top',end:'bottom top',scrub:true,invalidateOnRefresh:true}
  });
});

// isometric surface
function point(i,j,time,mouseX,mouseY){
  var cell=clamp(canvasWidth/19,42,78), originX=canvasWidth*.61, originY=canvasHeight*.34;
  var x=originX+(i-j)*cell*.92, y=originY+(i+j)*cell*.43;
  var wave=Math.sin(i*.72+j*.54+time*.00105)*cell*.18+Math.sin((i-j)*.48+time*.00072)*cell*.1;
  var distance=Math.hypot(x-mouseX,y-mouseY)||1;
  var pull=mouse.active?Math.exp(-(distance*distance)/(cell*cell*18)):0;
  return {x:x+((x-mouseX)/distance)*pull*cell*.36,y:y+wave-pull*cell*.74,
    alpha:clamp(.08+(y/canvasHeight)*.18+pull*.38,0,.62)};
}

function draw(time){
  context.clearRect(0,0,canvasWidth,canvasHeight);context.lineWidth=.72;
  for(var row=-8;row<14;row++){
    context.beginPath();
    for(var column=-8;column<14;column++){
      var current=point(row,column,time,mouse.x,mouse.y), next=point(row+1,column,time,mouse.x,mouse.y);
      context.moveTo(current.x,current.y);context.lineTo(next.x,next.y);
    }
    context.stroke();
  }
  requestAnimationFrame(draw);
}

function renderLeader(value){
  var progress=clamp01(value), toCatalog=smoothstep((progress-.18)/.20), toProjects=smoothstep((progress-.58)/.20);
  var reveals=[1,toCatalog,toProjects];
  shots.forEach(function(shot,index){
    var reveal=reveals[index], hidden=((1-reveal)*100).toFixed(3);
    shot.style.clipPath='inset(0 '+hidden+'% 0 0)';
    shot.style.transform='scale('+(1.035-reveal*.035).toFixed(4)+')';
    shot.style.opacity=reveal.toFixed(3);
  });
  progressLine.style.transform='scaleX('+progress.toFixed(4)+')';
  setStep(progress<.34?0:(progress<.70?1:2));
}

function activateServiceRow(index){
  serviceRows.forEach(function(row,rowIndex){row.classList.toggle('is-active',rowIndex===index);});
}

Коротко о главном

( FAQ )
Сколько стоит сайт?
Стоимость зависит от объёма и задач, поэтому не вывешиваю её на витрину. Начинаю с бесплатного разбора и макета, после них называю точную сумму и фиксирую её до старта. Лендинг дешевле, редизайн и многостраничник дороже, потому что работы больше.
Какие сроки?
Лендинг занимает примерно неделю, если не затягивать с материалами и ответами. Редизайн и корпоративный сайт обычно занимают 1-2 недели в зависимости от объёма.
Что если нет текстов и фото?
Это нормально. Помогу собрать структуру, приведу тексты в порядок и подскажу, какие фото сделать или заменить. Начнём с короткого брифа.
Как происходит оплата?
Обычно 50% предоплата для старта и 50% после готовности и вашей приёмки. Способ обсудим: на карту или по реквизитам.
Что после запуска?
Выкладываю сайт на домен, передаю доступы и остаюсь на связи. Мелочи по запуску не бросаю, а дальнейшие доработки можно обсудить отдельно.
